────────────────────────────────────────────────────────────────────────────────
API OVERVIEW
────────────────────────────────────────────────────────────────────────────────

useKeyboardShortcuts({
  enabled?: boolean;              // Default: true
  onEscape?: () => void;          // Callback for Escape
  onSaveShortcut?: () => void;    // Callback for Ctrl+S
  autoSave?: boolean;             // Auto-save without dialog
  defaultMissionName?: string;    // Name for auto-save
})

getShortcutHint(action: string): string
  - Returns platform-specific shortcut text
  - Example: "Ctrl+S" or "Cmd+S"

────────────────────────────────────────────────────────────────────────────────
USAGE EXAMPLE
────────────────────────────────────────────────────────────────────────────────

import { useKeyboardShortcuts } from './hooks';

function App() {
  const [showDialog, setShowDialog] = useState(false);

  useKeyboardShortcuts({
    enabled: true,
    onEscape: () => setShowDialog(false),
    onSaveShortcut: () => setShowDialog(true),
  });

  return <div>...</div>;
}
